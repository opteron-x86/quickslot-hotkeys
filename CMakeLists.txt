function(quickslot_svg_dependency)
  set(BUILD_SHARED_LIBS OFF)
  if(MSVC)
    set(MATH_LIBRARY "")
  endif()
  set(LUNASVG_BUILD_EXAMPLES OFF CACHE BOOL "Build SVG examples" FORCE)
  add_subdirectory("${QUICKSLOT_VENDOR_DIR}/lunasvg" "${CMAKE_CURRENT_BINARY_DIR}/lunasvg" EXCLUDE_FROM_ALL)
endfunction()
